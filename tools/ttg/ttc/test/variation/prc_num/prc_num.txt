━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントの位置付け
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントは，テストで使用するプロセッサ数が不足している場合の対
応が，TTC(TOPPERS Test checker)で要件通り正しく行われていることを確認す
るためのテスト仕様書である．

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
プロセッサ数に関する要件
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
(要件1)
事前に定義されたターゲットのプロセッサ数を超過している場合は，そのテス
トケースが除外されること．また，テスト対象外とされたテストケースがあっ
た場合に，実行結果として【備考】に示した情報が表示されること．

【備考】
・実行結果として表示される情報
  ・入力されたテストケースの総数
  ・実行可能であったテストケースの数
  ・NGテストケースとして除外されたテストケースの数
  ・NGテストケースとして除外されたテストケースのテストIDと除外理由(英語)
  ・総数と除外した数から，実行可能であったテストケースの割合(％) 

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
要件に対する確認内容
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
テストに用いたTESRYデータは，ttsp/api_test/配下にある全TESRYデータであ
る．
全TESRYデータの内，プロセッサが3個必要なTESRYデータは61件である．

----------------------------------------------------------------------
(要件1)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのprc_numに2を指定
 (もしくはTTG実行の際に，オプション"--prc_num 2"を指定)
・ttsp/api_test/配下にある全TESRYデータを引数に指定してTTG実行

【確認内容】
・TTGにより生成されたソースコードに，プロセッサが3個必要な61件のテスト
  ケースが除外されていることを確認した
・要件の【備考】に記載された各項目が出力されていることを確認した
 (NGテストケースとして除外されたテストケースのテストIDと除外理由は，除
  外されたテストケースのファイルパスが出力されるttg_exclusion_list.txt
  に記載されているファイルパスによって確認した)


以上．
