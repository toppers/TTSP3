━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントの位置付け
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントは，タイマアーキテクチャに関する対応が，TTC(TOPPERS 
Test checker)で要件通り正しく行われていることを確認するためのテスト仕様
書である．

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
タイマアーキテクチャに関する要件
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
(要件1)
configure.yamlでローカルタイマ方式を指定した場合，TESRYデータの
variation属性であるtimer_archにglobalと指定されたグローバルタイマ方式限
定のテストケースはテスト対象外とされること．また，テスト対象外とされた
テストケースがあった場合に，実行結果として【備考】に示した情報が表示さ
れること．

(要件2)
configure.yamlでグローバルタイマ方式を指定した場合，TESRYデータの
variation属性であるtimer_archにlocalと指定されたローカルタイマ方式限定
のテストケースはテスト対象外とされること．また，テスト対象外とされたテ
ストケースがあった場合に，実行結果として【備考】に示した情報が表示され
ること．

【備考】
・実行結果として表示される情報
  ・入力されたテストケースの総数
  ・実行可能であったテストケースの数
  ・NGテストケースとして除外されたテストケースの数
  ・NGテストケースとして除外されたテストケースのテストIDと除外理由(英語)
  ・総数と除外した数から，実行可能であったテストケースの割合(％)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
要件に対する確認内容
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
テストに用いたTESRYデータは，ttsp/api_test/配下にある全TESRYデータであ
る．
全TESRYデータの内，ローカルタイマ方式限定のTESRYデータは202件，グローバ
ルタイマ方式限定のTESRYデータは17件である．

----------------------------------------------------------------------
(要件1)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのtimer_archにlocalを指定
 (もしくはTTG実行の際に，オプション"--timer_arch local"を指定)
・ttsp/api_test/配下にある全TESRYデータを引数に指定してTTG実行

【確認内容】
・TTGにより生成されたソースコードに，グローバルタイマ方式限定の17件のテ
  ストケースが除外されていることを確認した
・要件の【備考】に記載された各項目が出力されていることを確認した
 (NGテストケースとして除外されたテストケースのテストIDと除外理由は，除
  外されたテストケースのファイルパスが出力されるttg_exclusion_list.txt
  に記載されているファイルパスによって確認した)

----------------------------------------------------------------------
(要件2)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのtimer_archにglobalを指定
 (もしくはTTG実行の際に，オプション"--timer_arch global"を指定)
・ttsp/api_test/配下にある全TESRYデータを引数に指定してTTG実行

【確認内容】
・TTGにより生成されたソースコードに，ローカルタイマ方式限定の202件のテ
  ストケースが除外されていることを確認した
・要件の【備考】に記載された各項目が出力されていることを確認した
 (NGテストケースとして除外されたテストケースのテストIDと除外理由は，除
  外されたテストケースのファイルパスが出力されるttg_exclusion_list.txt
  に記載されているファイルパスによって確認した)
