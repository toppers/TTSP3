━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントの位置付け
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントは，非タスクコンテキストからの割込み優先度マスク変更サ
ポートに関する対応が，TTC(TOPPERS Test checker)で要件通り正しく行われて
いることを確認するためのテスト仕様書である．

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
非タスクコンテキストからの割込み優先度マスク変更サポートに関する要件
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
(要件1)
configure.yamlでenable_chg_ipm_in_non_taskに対してfalseが指定された場合，
TESRYデータのvariation属性であるenable_chg_ipm_in_non_taskにtrueと指定
された非タスクコンテキストから割込み優先度マスクを変更するテストケース
は，テスト対象外とされること．また，テスト対象外とされたテストケースが
あった場合に，実行結果として【備考】に示した情報が表示されること．

(要件2)
configure.yamlでenable_chg_ipm_in_non_taskに対してtrueが指定された場合，
TESRYデータのvariation属性であるenable_chg_ipm_in_non_taskにtrueと指定
された非タスクコンテキストから割込み優先度マスクを変更するテストケース
でも，テスト対象とされること．

【備考】
・実行結果として表示される情報
  ・入力されたテストケースの総数
  ・実行可能であったテストケースの数
  ・NGテストケースとして除外されたテストケースの数
  ・NGテストケースとして除外されたテストケースのテストIDと除外理由(英語)
  ・総数と除外した数から，実行可能であったテストケースの割合(％)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
要件に対する確認内容
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
テストに用いたTESRYデータの概要を以下にまとめる(本ドキュメントと同フォ
ルダに格納されている)．
以下のTESRYデータは，要件を満たしていることを確認するためのTESRYデータ
であるため，APIを使用せず割込み優先度マスクを変更するTESRYデータではな
い．
・enable_chg_ipm_in_non_task_true.yaml
  ・variation属性であるenable_chg_ipm_in_non_taskにtrueを指定
・enable_chg_ipm_in_non_task_false.yaml
  ・variation属性であるenable_chg_ipm_in_non_taskにfalseを指定

----------------------------------------------------------------------
(要件1)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのenable_chg_ipm_in_non_taskにfalseを指定
 (もしくはTTG実行の際に，オプション"--enable_chg_ipm_in_non_task false"
  を指定)
・以下のTESRYデータを引数に指定してTTG実行
  ・enable_chg_ipm_in_non_task_true.yaml
  ・enable_chg_ipm_in_non_task_false.yaml

【確認内容】
・TTGにより生成されたソースコードに，
  enable_chg_ipm_in_non_task_true.yamlが除外されていることを確認した
・要件の【備考】に記載された各項目が出力されていることを確認した
 (NGテストケースとして除外されたテストケースのテストIDと除外理由は，除
  外されたテストケースのファイルパスが出力されるttg_exclusion_list.txt
  に記載されているファイルパスによって確認した)

----------------------------------------------------------------------
(要件2)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのenable_chg_ipm_in_non_taskにtrueを指定
 (もしくはTTG実行の際に，オプション"--enable_chg_ipm_in_non_task true"
  を指定)
・以下のTESRYデータを引数に指定してTTG実行
  ・enable_chg_ipm_in_non_task_true.yaml
  ・enable_chg_ipm_in_non_task_false.yaml

【確認内容】
・TTGにより生成されたソースコードに，除外されたテストケースが無いことを
  確認した
