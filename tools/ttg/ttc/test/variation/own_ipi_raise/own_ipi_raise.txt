━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントの位置付け
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
このドキュメントは，pre_conditionでの過渡状態サポートに関する対応が，
TTC(TOPPERS Test checker)で要件通り正しく行われていることを確認するため
のテスト仕様書である．

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
過渡状態サポートに関する要件
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
(要件1)
configure.yamlでown_ipi_raiseに対してfalseが指定された場合，
pre_conditionで過渡状態のタスクが登場するテストケースは，テスト対象外と
されること．また，テスト対象外とされたテストケースがあった場合に，実行
結果として【備考】に示した情報が表示されること．

(要件2)
configure.yamlでown_ipi_raiseに対してfalseが指定された場合，
post_conditionのみに過渡状態のタスクが登場するテストケースは，テスト対
象とされること．

(要件3)
configure.yamlでown_ipi_raiseに対してtrueが指定された場合，
pre_conditionで過渡状態のタスクが登場するテストケースでも，テスト対象と
されること．

【備考】
・実行結果として表示される情報
  ・入力されたテストケースの総数
  ・実行可能であったテストケースの数
  ・NGテストケースとして除外されたテストケースの数
  ・NGテストケースとして除外されたテストケースのテストIDと除外理由(英語)
  ・総数と除外した数から，実行可能であったテストケースの割合(％)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
要件に対する確認内容
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
テストに用いたTESRYデータは，ttsp/api_test/FMP/配下にある全TESRYデータ
である．
全TESRYデータの内，pre_conditionで過渡状態のタスクが登場するTESRYデータ
は204件，doでsus_tsk()を発行することによりpost_conditionで過渡状態のタ
スクが登場するTESRYデータは5件である．

----------------------------------------------------------------------
(要件1)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのown_ipi_raiseにfalseを指定
 (もしくはTTG実行の際に，オプション"--own_ipi_raise false"を指定)
・ttsp/api_test/FMP/配下にある全TESRYデータを引数に指定してTTG実行

【確認内容】
・TTGにより生成されたソースコードに，pre_conditionで過渡状態のタスクが
  登場するテストケースが除外されていることを確認した
・要件の【備考】に記載された各項目が出力されていることを確認した
 (NGテストケースとして除外されたテストケースのテストIDと除外理由は，除
  外されたテストケースのファイルパスが出力されるttg_exclusion_list.txt
  に記載されているファイルパスによって確認した)

----------------------------------------------------------------------
(要件2)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのown_ipi_raiseにfalseを指定
 (もしくはTTG実行の際に，オプション"--own_ipi_raise false"を指定)
・ttsp/api_test/FMP/配下にある全TESRYデータを引数に指定してTTG実行

【確認内容】
・TTGにより生成されたソースコードに，post_conditionで過渡状態のタスクが
  登場するテストケースが含まれていることを確認した

----------------------------------------------------------------------
(要件3)に対する確認
----------------------------------------------------------------------
【実施内容】
・configure.yamlのown_ipi_raiseにtrueを指定
 (もしくはTTG実行の際に，オプション"--own_ipi_raise true"を指定)
・ttsp/api_test/FMP/配下にある全TESRYデータを引数に指定してTTG実行

【確認内容】
・TTGにより生成されたソースコードに，pre_conditionで過渡状態のタスクが
  登場するテストケースが含まれていることを確認した
